# Native programs (llrender/llvknative.h). One line each:
#   override | <name of the viewer program it stands in for> | <vertex file> | <fragment file> | <family> [| define NAME=VALUE] [| spec ID=VALUE]
# The family is the live key native_<family>=1 in rt-debug.txt (0 or missing: the old program). native_reload=0 turns the watching of the files off.
override | Copy Shader | copy.vert | copy.frag | copy | default on
override | Deferred Blur Light Shader | fullscreen.vert | blur_light.frag | blur | default on
override | Deferred Sun Shader | fullscreen.vert | sun_lightmap.frag | sun | require ssao_radius | require ll_ti_shadowMap0 | default on
# family 2: the transparent surfaces (doc/vulkan/SHADERS-F2-alpha.md). Fullbright first (rt-debug.txt alphauber=0 keeps the fullbright draws in their own program); native_alpha=1 turns it on.
override | Deferred Fullbright Alpha Pull Shader | alpha.vert | alpha_fullbright.frag | alpha | default on
override | Deferred Fullbright Alpha Pull Skin Shader | alpha.vert | alpha_fullbright.frag | alpha | define HAS_SKIN=1 | define HAS_INDEXED_DRAW=1 | default on
# the lit case (plain diffuse), two variants of each program by the sky: classic_mode=1 (small: the ambient light is the sky's) and classic_mode=0 (ALPHA_PROBES: the ambient light comes from the reflection probes)
override | Deferred Alpha Pull Shader | alpha.vert | alpha_lit.frag | alpha | define ALPHA_LIT=1 | require shadow_matrix | when classic_mode=1 | default on
override | Deferred Alpha Pull Shader | alpha.vert | alpha_lit.frag | alpha | define ALPHA_LIT=1 | define ALPHA_PROBES=1 | require shadow_matrix | when classic_mode=0 | default on
override | Deferred Alpha Pull Skin Shader | alpha.vert | alpha_lit.frag | alpha | define ALPHA_LIT=1 | define HAS_SKIN=1 | define HAS_INDEXED_DRAW=1 | require shadow_matrix | when classic_mode=1 | default on
override | Deferred Alpha Pull Skin Shader | alpha.vert | alpha_lit.frag | alpha | define ALPHA_LIT=1 | define ALPHA_PROBES=1 | define HAS_SKIN=1 | define HAS_INDEXED_DRAW=1 | require shadow_matrix | when classic_mode=0 | default on
# the merged lit + fullbright program (the row's fullbright flag picks the branch): the program the alpha runs use when alphamat is 0 (the material draws have their own programs then)
override | Deferred Alpha Uber Pull Shader | alpha.vert | alpha_lit.frag | alpha | define ALPHA_LIT=1 | define ALPHA_UBER=1 | require shadow_matrix | when classic_mode=1 | default on
override | Deferred Alpha Uber Pull Shader | alpha.vert | alpha_lit.frag | alpha | define ALPHA_LIT=1 | define ALPHA_PROBES=1 | define ALPHA_UBER=1 | require shadow_matrix | when classic_mode=0 | default on
override | Deferred Alpha Uber Pull Skin Shader | alpha.vert | alpha_lit.frag | alpha | define ALPHA_LIT=1 | define ALPHA_UBER=1 | define HAS_SKIN=1 | define HAS_INDEXED_DRAW=1 | require shadow_matrix | when classic_mode=1 | default on
override | Deferred Alpha Uber Pull Skin Shader | alpha.vert | alpha_lit.frag | alpha | define ALPHA_LIT=1 | define ALPHA_PROBES=1 | define ALPHA_UBER=1 | define HAS_SKIN=1 | define HAS_INDEXED_DRAW=1 | require shadow_matrix | when classic_mode=0 | default on
# the merged program with the material draws too (the program the alpha runs use by default): fullbright, lit and material draws in one native program; the probes are used by the material path in every sky
override | Deferred Alpha Uber Material Pull Shader | alpha.vert | alpha_lit.frag | alpha | define ALPHA_LIT=1 | define ALPHA_PROBES=1 | define ALPHA_UBER=1 | define ALPHA_MAT=1 | require shadow_matrix | default on
override | Deferred Alpha Uber Material Pull Skin Shader | alpha.vert | alpha_lit.frag | alpha | define ALPHA_LIT=1 | define ALPHA_PROBES=1 | define ALPHA_UBER=1 | define ALPHA_MAT=1 | define HAS_SKIN=1 | define HAS_INDEXED_DRAW=1 | require shadow_matrix | default on
